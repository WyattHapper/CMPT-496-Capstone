// Workflow: Table Creation and Data Ingestion
// Description: Handles the initialization of console tables from various data sources while enforcing structural constraints on columns and rows.
// Covered Business Rules: [2, 3, 5, 9]

[Fact]
public void TableCreationAndDataIngestionWorkflow_ShouldSucceedForAllSourcesAndEnforceConstraints()
{
    // 1. Verify Ingestion from POCOs (Rule 2 & 9)
    var users = new List<User>
    {
        new() { Name = "Alexandre", Age = 36 },
        new() { Name = "René", Age = 59 }
    };
    var pocoTable = ConsoleTable.From(users);
    Assert.Equal(2, pocoTable.Rows.Count);
    Assert.Contains("Name", pocoTable.Columns.Cast<string>());
    Assert.Contains("Age", pocoTable.Columns.Cast<string>());

    // 2. Verify Ingestion from Dictionaries (Rule 5 & 9)
    var dictData = new Dictionary<string, Dictionary<string, object>>
    {
        { "Row1", new Dictionary<string, object> { { "ColA", "Val1" }, { "ColB", 100 } } },
        { "Row2", new Dictionary<string, object> { { "ColA", "Val2" }, { "ColB", 200 } } }
    };
    var dictTable = ConsoleTable.FromDictionary(dictData);
    Assert.Equal(2, dictTable.Rows.Count);
    // FromDictionary adds an empty string as the first column for row labels
    Assert.Equal("", dictTable.Columns[0].ToString()); 
    Assert.Equal("Row1", dictTable.Rows[0][0]);

    // 3. Verify Ingestion from DataTable (Rule 9)
    var dataTable = new DataTable();
    dataTable.Columns.Add("ID", typeof(int));
    dataTable.Columns.Add("Status", typeof(string));
    dataTable.Rows.Add(1, "Active");
    var dtTable = ConsoleTable.From(dataTable);
    Assert.Single(dtTable.Rows);
    Assert.Equal("ID", dtTable.Columns[0]);

    // 4. Verify Ingestion from Object Array (Rule 9)
    Object[][] arrayData = 
    [
        ["Header1", "Header2"],
        ["Data1", "Data2"]
    ];
    var arrayTable = ConsoleTable.From(arrayData);
    Assert.Single(arrayTable.Rows);
    Assert.Equal("Header1", arrayTable.Columns[0]);

    // 5. Verify Structural Constraints (Rule 3)
    // Constraint: Columns must be defined prior to row addition
    var emptyTable = new ConsoleTable();
    var colException = Assert.Throws<Exception>(() => emptyTable.AddRow(1, 2, 3));
    Assert.Equal("Please set the columns first", colException.Message);

    // Constraint: Every row must match defined column count
    var constrainedTable = new ConsoleTable("Col1", "Col2");
    var countException = Assert.Throws<Exception>(() => constrainedTable.AddRow("OnlyOneItem"));
    Assert.Contains("does not match the values", countException.Message);

    // Verify successful manual ingestion after correct setup
    constrainedTable.AddRow("Value1", "Value2");
    Assert.Single(constrainedTable.Rows);
}

// Workflow: Visual Layout and Alignment Calculation
// Description: Calculates column widths and alignment policies based on data types and Unicode character properties to ensure correct visual layout.
// Covered Business Rules: [4, 7, 8]

[Fact]
public void VisualLayoutAndAlignmentCalculation_IntegrationTest()
{
    // 1. Arrange - Prepare source data using existing User model from context
    var users = new List<User>
    {
        new User { Name = "Alexandre", Age = 36 }
    };

    // 2. Act - Build table from existing schema and configure layout policies
    var table = ConsoleTable.From(users)
        .Configure(o => o.NumberAlignment = Alignment.Right);

    // Rule 4 & 8: Set visual width and wrapping constraints
    // MaxWidth is measured in string character length before the UnicodeCalculator calculates actual visual width
    table.MaxWidth = 10;
    table.SetWordBreakDelimiter(' ');

    // Add a row with Unicode characters to verify visual width calculation (Rule 8)
    // "Unicode 哈哈" string length is 10, visual width is 12
    table.AddRow("Unicode 哈哈", 5000);

    // Add a row that exceeds MaxWidth to trigger wrapping (Rule 4)
    // "Wrapping Test Data" length is 18 characters, exceeds MaxWidth 10
    table.AddRow("Wrapping Test Data", 7);

    var output = table.ToString();

    // 3. Assert - Verify alignment and visual formatting

    // Rule 7 & 8: Verify Right-Alignment for Numeric Types (Age column)
    // The Age column width should be at least 5 to accommodate "5000" (length 4) plus spacing
    // "36" should be padded with leading spaces for right-alignment
    Assert.Contains("| Alexandre  |   36 |", output);
    Assert.Contains("|   5000 |", output);
    Assert.Contains("|      7 |", output);

    // Rule 8: Verify Unicode Visual Width handling
    // The Name column header is "Name" (4). "Alexandre" (9). "Unicode 哈哈" (12 visual length)
    // ColumnLengths() uses UnicodeCalculator.GetWidth to find the max visual width
    Assert.Contains("| Unicode 哈哈 |", output);

    // Rule 4: Verify Wrapping at MaxWidth using WordBreakDelimiter
    // "Wrapping Test Data" should be broken at the space character
    // "Wrapping " fits under 10 chars. "Test Data" is pushed to the next row.
    // Since the logic inserts a new row, we check for segments
    Assert.Contains("| Wrapping   |", output);
    Assert.Contains("| Test Data  |", output);
}

// Workflow: Output Rendering and Formatting
// Description: Generates the final formatted string output in various styles and sends it to the target output stream, managing optional elements like row counts.
// Covered Business Rules: [1, 6, 10, 11]

[Fact] public void VerifyOutputRenderingAndFormattingWorkflow() { var users = new List<User> { new User { Name = "Alexandre", Age = 36 }, new User { Name = "René", Age = 59 } }; var stringWriter = new StringWriter(); var table = ConsoleTable.From(users); Assert.Equal(Console.Out, table.Options.OutputTo); table.Configure(o => { o.OutputTo = stringWriter; o.NumberAlignment = Alignment.Right; o.EnableCount = true; }); table.Write(Format.Default); string defaultOutput = stringWriter.ToString(); Assert.Contains("Alexandre", defaultOutput); Assert.Contains("Count: 2", defaultOutput); Assert.Contains(" 36 |", defaultOutput); stringWriter.GetStringBuilder().Clear(); table.Configure(o => o.EnableCount = false); table.Write(Format.Minimal); string minimalOutput = stringWriter.ToString(); Assert.Contains("René", minimalOutput); Assert.DoesNotContain("Count: 2", minimalOutput); stringWriter.GetStringBuilder().Clear(); table.Write(Format.MarkDown); string markdownOutput = stringWriter.ToString(); Assert.Contains("| Name | Age |", markdownOutput); Assert.Contains("|---|---", markdownOutput); stringWriter.GetStringBuilder().Clear(); table.Write(Format.Alternative); string alternativeOutput = stringWriter.ToString(); Assert.Contains("+---", alternativeOutput); Assert.Contains("| Alexandre", alternativeOutput); }
