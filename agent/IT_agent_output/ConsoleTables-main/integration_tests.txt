Rule 3: Verify rows cannot be added before columns are defined\n    var emptyTable = new ConsoleTable();\n    var emptyEx = Assert.Throws<Exception>(() => emptyTable.AddRow(1, 2));\n    Assert.Equal(\"Please set the columns first\", emptyEx.Message);\n\n    // 2. // Workflow: Table Visual Styling and Configuration
// Description: Defines the visual presentation and formatting capabilities of the table, including predefined themes, Unicode-aware alignment, and data-specific positioning.
// Covered Business Rules: [2, 5, 7, 10]

System
System.Collections.Generic
System.Linq
System.Text
Xunit
ConsoleTables

[Fact]
public void VerifyTableVisualStylingAndConfigurationWorkflow()
{
    // 1. Setup data representing a mix of types and Unicode characters
    var users = new List<User>
    {
        new User { Name = "Alexandre 哈哈", Age = 36 },
        new User { Name = "René", Age = 59 }
    };

    // 2. Initialize the table from the collection and configure visual properties
    // Rule 2 & 7: Column alignment configuration for numeric data types
    var table = ConsoleTable
        .From(users)
        .Configure(o => 
        {
            o.NumberAlignment = Alignment.Right;
            o.EnableCount = true;
        });

    // 3. Verify Default Boxed Output (Rule 2, 5, 10)
    // Testing Unicode-aware width calculation and right-aligned numeric '36' and '59'
    var defaultOutput = table.ToString();
    Assert.Contains("Alexandre 哈哈", defaultOutput);
    Assert.Contains("|  36 |", defaultOutput); // Note the leading space for right alignment
    Assert.Contains("|  59 |", defaultOutput);
    Assert.Contains("Count: 2", defaultOutput);

    // 4. Verify Markdown Style (Rule 10)
    var markdownOutput = table.ToMarkDownString();
    Assert.StartsWith("| Name", markdownOutput.Trim());
    Assert.Contains("|-----------", markdownOutput);
    Assert.Contains("| Alexandre 哈哈 |  36 |", markdownOutput);

    // 5. Verify Minimal Style (Rule 2, 10)
    var minimalOutput = table.ToMinimalString();
    Assert.DoesNotContain("|", minimalOutput);
    Assert.Contains("Alexandre 哈哈", minimalOutput);
    Assert.Contains("36", minimalOutput);

    // 6. Verify Alternative Style (Rule 2)
    var alternativeOutput = table.ToStringAlternative();
    Assert.Contains("+", alternativeOutput);
    Assert.Contains("| Name", alternativeOutput);

    // 7. Verify interaction with Manual Row Addition and Unicode
    var manualTable = new ConsoleTable("Item", "Value")
        .AddRow("Total 總額", 1000)
        .Configure(o => o.NumberAlignment = Alignment.Right);

    var manualOutput = manualTable.ToString();
    
    // Check if Unicode string and right-aligned number co-exist correctly in formatted string
    Assert.Contains("Total 總額", manualOutput);
    Assert.Contains("| 1000 |", manualOutput);
}

// Workflow: Table Finalization and Output Execution
// Description: Manages the final rendering process, including cell content wrapping, the inclusion of optional summary footers, and directing the formatted string to the output stream.
// Covered Business Rules: [4, 6, 8, 12]

System.IO
Xunit
ConsoleTables
System.Linq

[Fact]
public void TableFinalizationAndOutputExecution_ShouldWrapContentAndHonorOptions()
{
    using (var testWriter = new StringWriter())
    {
        var options = new ConsoleTableOptions
        {
            Columns = new[] { "Data" },
            EnableCount = true,
            OutputTo = testWriter
        };
        var table = new ConsoleTable(options);
        table.MaxWidth = 10;
        table.AddRow("Wrapping Logic");
        table.Write();
        string output = testWriter.ToString();
        Assert.Contains("Wrapping", output);
        Assert.Contains("Logic", output);
        Assert.Contains("Count: 2", output);
        Assert.False(string.IsNullOrWhiteSpace(output));
    }
}
