// Workflow: Table Initialization and Data Population
// Description: Handles the creation of table structures and the ingestion of data from various sources such as POCOs, dictionaries, and arrays, while enforcing schema consistency.
// Covered Business Rules: [2, 3, 5, 9]

using System;
using System.Collections.Generic;
using System.Data;
using System.IO;
using System.Linq;
using Xunit;
using ConsoleTables;
using ConsoleTables.Tests;

[Fact]
public void TableInitializationAndDataPopulation_IntegrationTest()
{
    // 1. Verify Rule 2 & 9: Automated generation from POCO collections (Reflection)
    var users = new List<ConsoleTableTest.User>
    {
        new() { Name = "IntegrationTestUser", Age = 25 }
    };
    var pocoTable = ConsoleTable.From(users);
    Assert.NotNull(pocoTable);
    Assert.Contains("Name", pocoTable.Columns);
    Assert.Contains("Age", pocoTable.Columns);
    Assert.Single(pocoTable.Rows);

    // 2. Verify Rule 5 & 9: Table creation from nested dictionaries with keys as row labels
    var dictData = new Dictionary<string, Dictionary<string, object>>
    {
        {"RowAlpha", new Dictionary<string, object> { { "Col1", true }, { "Col2", 100 } } }
    };
    var dictTable = ConsoleTable.FromDictionary(dictData);
    Assert.Equal(3, dictTable.Columns.Count); // Empty label column + 2 keys
    Assert.Equal("", dictTable.Columns[0]);
    Assert.Equal("RowAlpha", dictTable.Rows[0][0]);

    // 3. Verify Rule 9: Table generation from Object Arrays and DataTables
    Object[][] arrayData = 
    [
        ["HeaderA", "HeaderB"],
        ["ValueA1", "ValueB1"]
    ];
    var arrayTable = ConsoleTable.From(arrayData);
    Assert.Equal("HeaderA", arrayTable.Columns[0].ToString());

    DataTable dataTable = new DataTable();
    dataTable.Columns.Add("DTCol");
    dataTable.Rows.Add("DTValue");
    var dtTable = ConsoleTable.From(dataTable);
    Assert.Equal("DTCol", dtTable.Columns[0]);

    // 4. Verify Rule 3: Enforcement of defined columns and item count consistency
    var manualTable = new ConsoleTable("Column1", "Column2");

    // Validate Rule 3: Row must match exactly 2 columns
    var countEx = Assert.Throws<Exception>(() => manualTable.AddRow("Mismatch"));
    Assert.Contains("does not match", countEx.Message);

    // Validate successful row addition
    manualTable.AddRow("Value1", "Value2");
    Assert.Single(manualTable.Rows);

    // Validate Rule 3: Cannot add row if no columns are defined
    var emptyOptions = new ConsoleTableOptions { Columns = new List<string>() };
    var headlessTable = new ConsoleTable(emptyOptions);
    var headlessEx = Assert.Throws<Exception>(() => headlessTable.AddRow("Data"));
    Assert.Equal("Please set the columns first", headlessEx.Message);
}

// Workflow: Visual Layout and Alignment Calculation
// Description: Determines the internal dimensions and alignment of the table by calculating Unicode character widths, word wrapping, and specific numeric alignment policies.
// Covered Business Rules: [4, 7, 8]

using System;
using System.Collections.Generic;
using System.Linq;
using Xunit;
using ConsoleTables;

[Fact]
public void VisualLayoutAndAlignment_FullWorkflow_ShouldMaintainVisualIntegrity()
{
    // Arrange
    // Business Rule 4, 7, 8: Define data with Unicode characters and numeric types
    // 'User' class is available from the test project context as seen in the source code
    var users = new List<ConsoleTables.Tests.User>
    {
        new ConsoleTables.Tests.User { Name = "User 哈哈", Age = 30 },
        new ConsoleTables.Tests.User { Name = "Wrap Candidate", Age = 1234 }
    };

    // Initialize the workflow via the fluent API
    var table = ConsoleTable.From(users);

    // Configure alignment and wrapping parameters
    // Rule 7 & 8: Support specific right-alignment for numeric data types
    table.Configure(options => 
    {
        options.NumberAlignment = Alignment.Right;
        options.EnableCount = true;
    });

    // Rule 4: Table rendering manages layout by wrapping and delimiters
    table.MaxWidth = 15;
    table.SetWordBreakDelimiter(' ');

    // Act
    // Trigger the layout calculation and formatting workflow
    var result = table.ToString();

    // Assert
    // Verify Rule 8: Unicode visual width calculation
    // "User 哈哈" has a visual width of 9 (5 English + 2*2 CJK).
    // Verify Rule 7: Numeric alignment logic (padded on the left for right alignment)

    // The Age column width should be 4 (based on "1234"). 
    // Format for numeric right alignment in ToString is " | {i, length}"
    // For Age 30: " |   30 |" (1 space from format template + 2 spaces padding 30 to width 4)
    // For Age 1234: " | 1234 |" (1 space from format template + 0 spaces padding 1234 to width 4)
    
    Assert.Contains("| User 哈哈", result);
    Assert.Contains("|   30 |", result);
    Assert.Contains("| 1234 |", result);

    // Verify numeric column header is also right-aligned per rule implementation
    // "Age" (3) in a column of 4: " |  Age |" (1 template space + 1 padding space)
    Assert.Contains("|  Age |", result);

    // Verify the workflow count configuration
    Assert.Contains("Count: 2", result);
}

// Workflow: Table Rendering and Output Generation
// Description: Produces the final visual output in various formats like Markdown or Minimal, including the configuration of row count summaries and target output streams.
// Covered Business Rules: [1, 6, 10, 11]

using System;
using System.IO;
using Xunit;
using ConsoleTables;

[Fact]
public void TableRenderingWorkflow_ShouldValidateOutputStylesAlignmentAndStreamRedirection()
{
    // 1. Verify Rule 10: Default output should be Console.Out
    var defaultTable = new ConsoleTable("DefaultCol");
    Assert.Equal(Console.Out, defaultTable.Options.OutputTo);

    // 2. Setup workflow components with custom stream and data
    var testWriter = new StringWriter();
    var table = new ConsoleTable("Item", "Quantity");
    table.AddRow("Widget A", 50);
    table.AddRow("Gadget B", 1000);

    // 3. Configure Workflow: Enable right alignment for numbers and disable row count (Rule 1 & 6)
    table.Configure(o =>
    {
        o.OutputTo = testWriter;
        o.NumberAlignment = Alignment.Right;
        o.EnableCount = false;
    });

    // 4. Exercise Workflow: Render as MarkDown (Rule 10)
    table.Write(Format.MarkDown);
    var markdownOutput = testWriter.ToString();

    // Validate MarkDown output and Right Alignment logic (Rule 1)
    Assert.Contains("| Item     | Quantity |", markdownOutput);
    Assert.Contains("| Widget A |       50 |", markdownOutput);
    Assert.Contains("| Gadget B |     1000 |", markdownOutput);
    // Validate exclusion of count (Rule 6)
    Assert.DoesNotContain("Count:", markdownOutput);

    // 5. Exercise Workflow: Render as Default with row count enabled (Rule 11)
    testWriter.GetStringBuilder().Clear();
    table.Configure(o => o.EnableCount = true);
    table.Write(Format.Default);
    var defaultOutput = testWriter.ToString();

    // Validate Default format borders and summary row count (Rule 11)
    Assert.Contains("----------------------", defaultOutput);
    Assert.Contains("| Item     | Quantity |", defaultOutput);
    Assert.Contains("Count: 2", defaultOutput);

    // 6. Exercise Workflow: Render as Minimal format (Rule 1 & 10)
    testWriter.GetStringBuilder().Clear();
    table.Write(Format.Minimal);
    var minimalOutput = testWriter.ToString();

    // Validate Minimal format specific rendering (No pipe delimiters)
    Assert.Contains("Item       Quantity", minimalOutput);
    Assert.DoesNotContain("|", minimalOutput);
}
