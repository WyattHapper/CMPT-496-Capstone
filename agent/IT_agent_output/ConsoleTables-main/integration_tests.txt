// Workflow: Table Ingestion and Data Sourcing
// Description: Processes and validates data from diverse sources such as POCOs, dictionaries, and lists to establish a structured table format.
// Covered Business Rules: [2, 3, 5, 9]

using System;
using System.Collections.Generic;
using System.Data;
using System.Linq;
using Xunit;
using ConsoleTables;

[Fact]
public void TableIngestionWorkflow_ShouldSupportMultipleSourcesAndValidateInput()
{
    // 1. POCO Source (Rule 2, Rule 9)
    // Uses the User class found in the provided source context
    var users = new List<User>
    {
        new User { Name = "Alice", Age = 30 },
        new User { Name = "Bob", Age = 25 }
    };
    var pocoTable = ConsoleTable.From(users);
    var pocoOutput = pocoTable.ToString();
    Assert.Contains("Alice", pocoOutput);
    Assert.Contains("Bob", pocoOutput);
    Assert.Contains("30", pocoOutput);

    // 2. Dictionary Source (Rule 5, Rule 9)
    // Rule 5: Keys serve as row labels (leftmost column)
    var dictData = new Dictionary<string, Dictionary<string, object>>
    {
        { "Row_1", new Dictionary<string, object> { { "Status", "Active" }, { "Code", 200 } } },
        { "Row_2", new Dictionary<string, object> { { "Status", "Pending" }, { "Code", 404 } } }
    };
    var dictTable = ConsoleTable.FromDictionary(dictData);
    var dictOutput = dictTable.ToString();
    Assert.Contains("Row_1", dictOutput);
    Assert.Contains("Row_2", dictOutput);
    Assert.Contains("Active", dictOutput);

    // 3. DataTable Source (Rule 9)
    var dataTable = new DataTable();
    dataTable.Columns.Add("HeaderA");
    dataTable.Columns.Add("HeaderB");
    dataTable.Rows.Add("Value1", "Value2");
    var dtTable = ConsoleTable.From(dataTable);
    Assert.Contains("HeaderA", dtTable.ToString());
    Assert.Contains("Value1", dtTable.ToString());

    // 4. Object Array Source (Rule 9)
    var arrayData = new object[][]
    {
        new object[] { "Column1", "Column2" },
        new object[] { "A1", "A2" },
        new object[] { "B1", "B2" }
    };
    var arrayTable = ConsoleTable.From(arrayData);
    Assert.Contains("Column1", arrayTable.ToString());
    Assert.Contains("A1", arrayTable.ToString());

    // 5. Validation Logic (Rule 3)
    // Rule 3: Columns must be defined first (done via constructor)
    var manualTable = new ConsoleTable("Col1", "Col2");
    
    // Rule 3: Row item count must exactly match the defined column count
    // Expecting exception when adding 3 items to a 2-column table
    var exceptionCount = Assert.Throws<Exception>(() => manualTable.AddRow(1, 2, 3));
    Assert.Contains("does not match", exceptionCount.Message);

    // Rule 3: Exception when adding rows before columns (tested via empty instance)
    var emptyTable = new ConsoleTable(new ConsoleTableOptions { Columns = new List<string>() });
    var exceptionNoColumns = Assert.Throws<Exception>(() => emptyTable.AddRow(1, 2));
    Assert.Equal("Please set the columns first", exceptionNoColumns.Message);
}

// Workflow: Dynamic Layout and Alignment Logic
// Description: Computes optimal column widths and handles text alignment by evaluating data types, Unicode character widths, and word-wrap constraints.
// Covered Business Rules: [4, 7, 8]

using ConsoleTables;
using System;
using System.Linq;
using Xunit;

[Fact]
public void DynamicLayoutAndAlignment_ShouldApplyWrappingAndNumericRightAlignment()
{
    // Arrange: Create a table with mixed data and alignment settings
    var table = new ConsoleTable("Product", "Qty");
    
    // Rule 4: Table rendering manages layout via MaxWidth and WordBreakDelimiter
    table.MaxWidth = 10;
    table.SetWordBreakDelimiter(' ');
    
    // Rule 7: Configure right-alignment for numeric types
    table.Configure(o => o.NumberAlignment = Alignment.Right);

    // Act:
    // Row 1: Content exceeding MaxWidth (11 chars) to trigger wrapping at WordBreakDelimiter (Rule 4)
    table.AddRow("Fresh Apples", 100);
    
    // Row 2: Unicode characters to verify visual width calculation (Rule 8)
    // "汉字" has 2 characters but visual width of 4
    table.AddRow("汉字", 50);

    var result = table.ToString();

    // Assert:
    // 1. Verify Word Wrapping (Rule 4)
    // "Fresh Apples" should wrap at the space into "Fresh" and "Apples"
    Assert.Contains("| Fresh      |", result);
    Assert.Contains("| Apples     |", result);

    // 2. Verify Right Alignment (Rule 7)
    // Column "Qty" width is 7 (header length). "100" (3 digits) should be right-aligned with leading spaces.
    // Composite format " | {1, 7} |" results in " |     100 |"
    Assert.Contains("|     100 |", result);
    Assert.Contains("|      50 |", result);

    // 3. Verify Unicode Visual Width (Rule 8)
    // The table structure should remain aligned despite the multi-byte characters.
    // "汉字" (width 4) + padding should match the total column width established by "Fresh Apples"
    Assert.Contains("| 汉字       |", result);
}

// Workflow: Multi-Format Output Presentation
// Description: Orchestrates the final rendering of the table into various visual styles and formats, managing summary footers and directing the results to the appropriate system output stream.
// Covered Business Rules: [1, 6, 10, 11]

using System;
using System.IO;
using Xunit;
using ConsoleTables;

[Fact]
public void MultiFormatOutputPresentation_ShouldApplyStylesAndManageFootersToOutputStream()
{
    // Arrange
    var testWriter = new StringWriter();
    var table = new ConsoleTable("Name", "Age");
    table.AddRow("Alice", 25);
    table.AddRow("Bob", 30);

    // Act & Assert - Step 1: Verify Default Format with Summary Count (Rule 1, 10, 11)
    table.Configure(o => {
        o.OutputTo = testWriter;
        o.EnableCount = true;
        o.NumberAlignment = Alignment.Right;
    });
    
    table.Write(Format.Default);
    var defaultOutput = testWriter.ToString();
    
    Assert.Contains("Alice", defaultOutput);
    Assert.Contains("Count: 2", defaultOutput);
    Assert.Contains("|", defaultOutput);

    // Act & Assert - Step 2: Verify Exclusion of Count Row (Rule 6)
    testWriter.GetStringBuilder().Clear();
    table.Configure(o => o.EnableCount = false);
    table.Write(Format.Default);
    var noCountOutput = testWriter.ToString();
    
    Assert.DoesNotContain("Count:", noCountOutput);

    // Act & Assert - Step 3: Verify Alternative Format (Rule 1)
    testWriter.GetStringBuilder().Clear();
    table.Write(Format.Alternative);
    var altOutput = testWriter.ToString();
    
    // Alternative format uses '+' and '-' for dividers
    Assert.Contains("+", altOutput);
    Assert.Contains("Alice", altOutput);

    // Act & Assert - Step 4: Verify Minimal Format (Rule 1, 10)
    testWriter.GetStringBuilder().Clear();
    table.Write(Format.Minimal);
    var minimalOutput = testWriter.ToString();
    
    // Minimal style typically has headers and content but fewer borders
    Assert.Contains("Alice", minimalOutput);
    Assert.DoesNotContain("|", minimalOutput);

    // Act & Assert - Step 5: Verify MarkDown Format (Rule 10)
    testWriter.GetStringBuilder().Clear();
    table.Write(Format.MarkDown);
    var markdownOutput = testWriter.ToString();
    
    Assert.Contains("|---", markdownOutput);
    Assert.Contains("Bob", markdownOutput);
}
