// Workflow: Visual Rendering and Data Transformation
// Description: Processes cell content through data type conversion and calculates layout dimensions using Unicode-aware width logic and alignment policies.
// Covered Business Rules: [4, 5, 8, 10, 13, 15]

using System;
using System.Data;
using Xunit;
using ConsoleTables;

[Fact]
public void Workflow_VisualRenderingAndDataTransformation_IntegrationTest()
{
    // Arrange: Create a DataTable to simulate complex input sources requiring data transformation (Rule 10).
    var dataTable = new DataTable();
    dataTable.Columns.Add("UnicodeLabel", typeof(string));
    dataTable.Columns.Add("NumericData", typeof(int));
    dataTable.Columns.Add("RawBytes", typeof(byte[]));

    // '你好' (Hello in Chinese) consists of 2 characters but has a visual width of 4.
    // Header 'UnicodeLabel' has a length of 12. Column width is determined by the header.
    // 123 (Numeric) should be right-aligned if configured (Rule 15).
    // Binary data { 10, 20, 30 } must be converted to a Base64 string for display (Rule 5).
    var binaryData = new byte[] { 10, 20, 30 };
    var expectedBase64 = Convert.ToBase64String(binaryData);
    dataTable.Rows.Add("你好", 123, binaryData);

    // Act: Load from DataTable, configure numeric alignment, and render (Rule 4, Rule 8).
    var table = ConsoleTable.From(dataTable)
        .Configure(options =>
        {
            options.NumberAlignment = Alignment.Right;
        });

    var output = table.ToMarkDownString();

    // Assert: Verify data type transformation for byte arrays (Rule 5, Rule 10).
    Assert.Contains(expectedBase64, output);

    // Assert: Verify visual width calculation and padding for Unicode (Rule 13, Rule 8).
    // Column 'UnicodeLabel' width is 12 (header length).
    // '你好' (width 4) in a column of 12 should have 8 fill spaces + padding.
    // In Markdown format: "| 你好         |" (1 space + 4 visual + 8 fill + 1 space)
    Assert.Contains("| 你好         |", output);

    // Assert: Verify numeric right-alignment configuration (Rule 4, Rule 15).
    // Column 'NumericData' width is 11 (header length).
    // '123' (width 3) right-aligned in width 11 should have 8 leading fill spaces.
    // In Markdown format: "|         123 |" (1 space + 8 fill + 3 content + 1 space)
    Assert.Contains("|         123 |", output);

    // Final check for the binary column alignment (header length 8).
    // Expected Base64 length: 4 ("ChMU"). Fill spaces: 4.
    Assert.Contains("| ChMU     |", output);
}

// Workflow: Presentation Style and Output Control
// Description: Manages the final visual style selection and directs the generated table output to specified console or file streams.
// Covered Business Rules: [2, 6, 9, 12, 16, 17]

using System;
using System.IO;
using System.Linq;
using Xunit;
using ConsoleTables;

[Fact]
public void PresentationStyleAndOutputControl_CompleteWorkflow_ShouldApplyConfigurationsAndRedirectOutput()
{
    // Arrange
    // Create a StringWriter to intercept the output stream (Rule 6, Rule 16)
    using var sw = new StringWriter();
    
    // Define columns and add rows including numeric types to verify alignment (Rule 2)
    var columns = new[] { "ID", "Label", "Value" };
    var table = new ConsoleTable(columns);
    table.AddRow(1, "Active", 100.5);
    table.AddRow(20, "Pending", 45.0);

    // Act: Configure the table options (Rule 2, Rule 12)
    table.Configure(o =>
    {
        o.OutputTo = sw;
        o.NumberAlignment = Alignment.Right;
        o.EnableCount = false; // Suppression of metadata (Rule 6)
    });

    // Step 1: Verify Markdown Format output (Rule 9, Rule 17)
    table.Write(Format.MarkDown);
    var markdownResult = sw.ToString();
    
    Assert.Contains("| ID | Label   | Value |", markdownResult);
    // ID column width 2. "1" is right-aligned: " 1"
    Assert.Contains("|  1 | Active  | 100.5 |", markdownResult);
    Assert.Contains("| 20 | Pending |  45.0 |", markdownResult);
    Assert.DoesNotContain("Count:", markdownResult); // Rule 12

    // Step 2: Clear stream and verify Minimal Format output (Rule 17)
    sw.GetStringBuilder().Clear();
    table.Write(Format.Minimal);
    var minimalResult = sw.ToString();
    
    Assert.NotNull(minimalResult);
    Assert.NotEmpty(minimalResult);
    // Minimal style typically lacks the border separators found in Default format
    Assert.DoesNotContain("----------------", minimalResult);
    Assert.DoesNotContain("Count:", minimalResult);

    // Step 3: Verify Alternative ASCII Format (Rule 9)
    sw.GetStringBuilder().Clear();
    table.Write(Format.Alternative);
    var alternativeResult = sw.ToString();
    
    // Alternative format typically uses '+' for intersection points
    Assert.Contains("+", alternativeResult);
    Assert.Contains("Active", alternativeResult);
    Assert.DoesNotContain("Count:", alternativeResult);

    // Final validation of the configuration persistence
    Assert.Equal(Alignment.Right, table.Options.NumberAlignment);
    Assert.False(table.Options.EnableCount);
    Assert.Same(sw, table.Options.OutputTo);
}
