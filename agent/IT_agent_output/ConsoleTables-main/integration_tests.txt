// Workflow: Data Ingestion and Integrity
// Description: Handles the initialization of tables and the populating of data from various sources like POCOs, dictionaries, and arrays while enforcing column-row count consistency.
// Covered Business Rules: [2, 3, 5, 9]

[Fact]
public void DataIngestionAndIntegrity_ShouldSupportMultipleSourcesAndEnforceConsistency()
{
    // 1. POCO Ingestion (Rule 2 & 9)
    // Uses a User POCO as found in the application source to verify automatic column discovery and data extraction
    var users = new List<User>
    {
        new User { Name = "Alexandre", Age = 36 }
    };
    var pocoTable = ConsoleTable.From(users);
    var pocoOutput = pocoTable.ToString();
    Assert.Contains("Name", pocoOutput);
    Assert.Contains("Age", pocoOutput);
    Assert.Contains("Alexandre", pocoOutput);
    Assert.Contains("36", pocoOutput);

    // 2. Dictionary Ingestion with Row Labels (Rule 5 & 9)
    // Verifies that dictionary keys are used as the first column (labels)
    var dictData = new Dictionary<string, Dictionary<string, object>>
    {
        {
            "Row_A", new Dictionary<string, object>
            {
                { "Col_1", true },
                { "Col_2", "Val_1" }
            }
        }
    };
    var dictTable = ConsoleTable.FromDictionary(dictData);
    var mdOutput = dictTable.ToMarkDownString();
    Assert.Contains("| Row_A | True  | Val_1 |", mdOutput);
    Assert.Contains("|   | Col_1 | Col_2 |", mdOutput);

    // 3. Object Array Ingestion (Rule 9)
    // Verifies the first array acts as headers and subsequent arrays as rows
    Object[][] arrayData = [
        ["H1", "H2"],
        ["D1", "D2"]
    ];
    var arrayTable = ConsoleTable.From(arrayData);
    Assert.Contains("H1", arrayTable.ToMinimalString());
    Assert.Contains("D1", arrayTable.ToMinimalString());

    // 4. DataTable Ingestion (Rule 9)
    var dt = new DataTable();
    dt.Columns.Add("Status", typeof(string));
    dt.Rows.Add("Active");
    var dtTable = ConsoleTable.From(dt);
    Assert.Contains("Status", dtTable.ToString());
    Assert.Contains("Active", dtTable.ToString());

    // 5. Enforce Consistency: Row addition constraints (Rule 3)
    
    // Case A: Row added before columns are defined
    var emptyTable = new ConsoleTable();
    var exNoColumns = Assert.Throws<Exception>(() => emptyTable.AddRow("Test"));
    Assert.Equal("Please set the columns first", exNoColumns.Message);

    // Case B: Row item count does not match column count
    var manualTable = new ConsoleTable("Col1", "Col2");
    var exMismatch = Assert.Throws<Exception>(() => manualTable.AddRow("OnlyOneVal"));
    Assert.Contains("The number columns in the row (2) does not match the values (1)", exMismatch.Message);

    // Case C: Valid row addition after columns are set
    manualTable.AddRow("V1", "V2");
    Assert.Single(manualTable.Rows);
    Assert.Equal(2, manualTable.Columns.Count);
}

// Workflow: Layout Calculation and Content Alignment
// Description: Manages the internal layout and visual alignment of table content by accounting for Unicode character widths, numeric data policies, and text wrapping constraints.
// Covered Business Rules: [4, 7, 8]

[Fact]
public void LayoutCalculationAndContentAlignment_ShouldHandleUnicodeAlignmentAndWrapping()
{
    // Arrange: Create a data set using the existing User class to test numeric alignment and text handling
    var users = new List<User>
    {
        // Rule 8: Unicode characters should be accounted for in visual width
        new User { Name = "CJK 哈哈", Age = 28 },
        // Rule 4: Long text should wrap based on MaxWidth at WordBreakDelimiters
        new User { Name = "ThisIsALongNameThatShouldWrapAtSpaces", Age = 45 }
    };

    // Rule 7: Configure the table to support right-alignment for numeric data types (Age)
    var table = ConsoleTable.From(users)
        .Configure(o => o.NumberAlignment = Alignment.Right);

    // Rule 4: Set wrapping constraints and a custom word break delimiter
    table.MaxWidth = 15;
    table.SetWordBreakDelimiter(' ');

    // Add a manual row to test wrapping at the custom delimiter
    table.AddRow("Space Wrap Test", 999);

    // Act: Render the table to a string
    var output = table.ToString();

    // Assert: Verify Right-Alignment for numeric Age column (Rule 7)
    // Age 28 should be right-padded/aligned inside the column format
    Assert.Contains("|  28 |", output);
    Assert.Contains("|  45 |", output);
    Assert.Contains("| 999 |", output);

    // Assert: Verify Unicode Width Calculation (Rule 8)
    // "CJK 哈哈" has 4 normal chars and 2 CJK chars (width 2 each). 
    // Total visual width should be 8. The layout must remain unbroken.
    Assert.Contains("| CJK 哈哈        |", output);

    // Assert: Verify Word Wrapping logic (Rule 4)
    // "Space Wrap Test" with MaxWidth 15 should break at the space
    // "Space" is 5 chars, "Wrap" is 4, "Test" is 4. 
    // "Space Wrap" is 10 chars, which fits in 15.
    Assert.Contains("| Space Wrap    |", output);
    Assert.Contains("| Test          |", output);

    // "ThisIsALongNameThatShouldWrapAtSpaces" has no spaces, 
    // so it should be forced to wrap at MaxWidth (15)
    Assert.Contains("| ThisIsALongNam |", output);
    Assert.Contains("| eThatShouldWra |", output);
    Assert.Contains("| pAtSpaces      |", output);
}

// Workflow: Visual Styling and Output Rendering
// Description: Applies visual styles and output formats—such as Markdown or Minimal—and manages the rendering of table footers and summary row counts to the output stream.
// Covered Business Rules: [1, 6, 10, 11]

[Fact]
public void VisualStylingAndOutputRenderingWorkflowTest()
{
    // Arrange: Create test data using the User model existing in the codebase
    var users = new List<User>
    {
        new User { Name = "Alice", Age = 25 },
        new User { Name = "Bob", Age = 30 }
    };
    
    // Arrange: Use StringWriter to capture output for validation (Rule 10)
    using var writer = new StringWriter();

    // Act: Configure table with specific alignment and styling (Rule 1, 11)
    var table = ConsoleTable.From(users)
        .Configure(o => 
        {
            o.OutputTo = writer;
            o.NumberAlignment = Alignment.Right;
            o.EnableCount = true;
        });

    // Step 1: Verify Default Format with Count enabled
    table.Write(Format.Default);
    var defaultOutput = writer.ToString();
    
    Assert.Contains("Name", defaultOutput);
    Assert.Contains("Age", defaultOutput);
    // Verify right alignment for numeric column (Rule 1)
    Assert.Contains("|  25 |", defaultOutput);
    // Verify row count footer (Rule 11)
    Assert.Contains("Count: 2", defaultOutput);

    // Step 2: Verify MarkDown Format (Rule 10)
    writer.GetStringBuilder().Clear();
    table.Write(Format.MarkDown);
    var markdownOutput = writer.ToString();
    
    Assert.Contains("| Name      | Age |", markdownOutput);
    Assert.Contains("|-----------|-----", markdownOutput);
    Assert.Contains("| Alice     |  25 |", markdownOutput);

    // Step 3: Verify Minimal Format with Count disabled (Rule 6)
    writer.GetStringBuilder().Clear();
    table.Configure(o => o.EnableCount = false);
    table.Write(Format.Minimal);
    var minimalOutput = writer.ToString();

    // Verify headers and content are present
    Assert.Contains("Name", minimalOutput);
    Assert.Contains("Alice", minimalOutput);
    // Verify total row count is excluded (Rule 6)
    Assert.DoesNotContain("Count: 2", minimalOutput);

    // Step 4: Verify Alternative Format (Rule 1)
    writer.GetStringBuilder().Clear();
    table.Write(Format.Alternative);
    var alternativeOutput = writer.ToString();

    // Verify the Alternative format uses + symbols for dividers
    Assert.Contains("+", alternativeOutput);
    Assert.Contains("Alice", alternativeOutput);
}
