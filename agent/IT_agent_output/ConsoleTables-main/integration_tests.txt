// Workflow: Table Data Ingestion and Source Processing
// Description: Handles the extraction of data from diverse sources such as POCOs, dictionaries, and standard lists while enforcing structural integrity between columns and rows.
// Covered Business Rules: [2, 3, 5, 9]

using System;
using System.Collections.Generic;
using System.Data;
using System.Linq;
using Xunit;
using ConsoleTables;

[Fact]
public void TableDataIngestion_ShouldProcessMultipleSourcesAndEnforceIntegrity()
{
    // 1. Process POCOs (Rule 2: Automatic generation from POCOs)
    var users = new List<User>
    {
        new User { Name = "Alice", Age = 30 },
        new User { Name = "Bob", Age = 25 }
    };
    var pocoTable = ConsoleTable.From(users);
    Assert.Equal(2, pocoTable.Rows.Count);
    Assert.Contains("Name", pocoTable.Columns.Cast<string>());
    Assert.Contains("Age", pocoTable.Columns.Cast<string>());

    // 2. Process Dictionaries (Rule 5: Keys serve as row labels)
    var dictData = new Dictionary<string, Dictionary<string, object>>
    {
        ["RowA"] = new Dictionary<string, object> { ["Col1"] = true, ["Col2"] = "Test" },
        ["RowB"] = new Dictionary<string, object> { ["Col1"] = false, ["Col2"] = "Demo" }
    };
    var dictTable = ConsoleTable.FromDictionary(dictData);
    Assert.Equal(2, dictTable.Rows.Count);
    // First column is the row label (empty string header), followed by Col1 and Col2
    Assert.Equal(3, dictTable.Columns.Count);
    Assert.Equal("RowA", dictTable.Rows[0][0]);

    // 3. Process DataTable and Object Arrays (Rule 9: Support multiple data sources)
    var dt = new DataTable();
    dt.Columns.Add("ID", typeof(int));
    dt.Columns.Add("Val", typeof(string));
    dt.Rows.Add(1, "Active");
    var dtTable = ConsoleTable.From(dt);
    Assert.Single(dtTable.Rows);
    Assert.Equal("1", dtTable.Rows[0][0]);

    object[][] arrayData = 
    [
        ["H1", "H2"],
        ["Data1", 100],
        ["Data2", 200]
    ];
    var arrayTable = ConsoleTable.From(arrayData);
    Assert.Equal(2, arrayTable.Rows.Count);
    Assert.Equal(2, arrayTable.Columns.Count);

    // 4. Enforce Structural Integrity (Rule 3: Columns first, then matching row count)
    var manualTable = new ConsoleTable("ColumnA", "ColumnB");
    
    // Valid addition
    manualTable.AddRow("Value1", "Value2");

    // Verify exception when row count does not match column count
    var rowMismatchEx = Assert.Throws<Exception>(() => manualTable.AddRow("OnlyOneValue"));
    Assert.Contains("does not match the values", rowMismatchEx.Message);

    // Verify exception when columns are not set
    var emptyTable = new ConsoleTable(new ConsoleTableOptions { Columns = new List<string>() });
    var noColumnsEx = Assert.Throws<Exception>(() => emptyTable.AddRow("Value"));
    Assert.Equal("Please set the columns first", noColumnsEx.Message);
}

// Workflow: Visual Layout and Alignment Logic
// Description: Determines the internal dimensions and alignment of the table by calculating Unicode character widths, handling word-wrapping, and applying numeric data policies.
// Covered Business Rules: [4, 7, 8]

using System;
using System.Collections.Generic;
using System.Linq;
using Xunit;
using ConsoleTables;
using ConsoleTables.Tests;

[Fact]
public void VisualLayoutWorkflow_ShouldHandleWrappingUnicodeAndNumericAlignment()
{
    // Arrange
    // Use the User entity found in the source context for the data source
    var users = new List<User>
    {
        // Name triggers wrapping at MaxWidth 15
        // Age is numeric, triggering right-alignment Rule 7
        new User { Name = "Alice With A Very Long Name That Wraps", Age = 25 },
        // Name contains Unicode characters (visual width 4) for Rule 8
        new User { Name = "哈哈", Age = 30 }
    };

    var table = ConsoleTable.From(users);
    
    // Configure workflow parameters
    table.MaxWidth = 15; 
    table.SetWordBreakDelimiter(' ');
    table.Configure(o => 
    {
        o.NumberAlignment = Alignment.Right;
        o.EnableCount = true;
    });

    // Act
    var output = table.ToString();

    // Assert
    // 1. Verify Numeric Right Alignment (Rule 7 & 8)
    // The Age column width will be based on the header "Age" (3) or content.
    // Since Age is right-aligned, "25" should have leading spaces in a fixed-width cell.
    // Composite format: " | {Age, 3} "
    Assert.Contains("|  25 |", output);
    Assert.Contains("|  30 |", output);

    // 2. Verify Unicode Visual Width Calculation (Rule 8)
    // "哈哈" visual width is 4. In a column of width 15, padding should account for this.
    // If visual width was calculated as 2 (char count), alignment would be off.
    Assert.Contains("哈哈", output);

    // 3. Verify Word Wrapping and MaxWidth logic (Rule 4)
    // "Alice With A Very Long Name That Wraps" length is 39.
    // MaxWidth is 15. The word break logic should split at spaces.
    // "Alice With A " (13 chars) - fits in 15.
    // "Very Long Name " (15 chars) - fits in 15.
    // "That Wraps" (10 chars) - fits in 15.
    Assert.Contains("Alice With A", output);
    Assert.Contains("Very Long Name", output);
    Assert.Contains("That Wraps", output);

    // 4. Verify vertical layout integrity (inserted rows for wrapping)
    // Check if the count reflects the number of logical users, not physical rows
    Assert.Contains("Count: 2", output);
}

// Workflow: Rendering Styles and Output Control
// Description: Manages the final visual representation of the data, including style selection, summary footer visibility, and routing the formatted table to the system output.
// Covered Business Rules: [1, 6, 10, 11]

using System;
using System.IO;
using System.Collections.Generic;
using Xunit;
using ConsoleTables;

[Fact]
public void RenderingStylesAndOutputControl_CompleteWorkflow()
{
    // 1. Verify Default Component Behavior (Rule 10: System Standard Output)
    var initialTable = new ConsoleTable("DefaultCheck");
    Assert.Equal(Console.Out, initialTable.Options.OutputTo);

    // 2. Setup Integration Environment with Custom Routing (Rule 10)
    using var testWriter = new StringWriter();
    var table = new ConsoleTable("Item", "Quantity");
    
    // 3. Configure Visual Styles and Footer (Rule 1: Alignment, Rule 11: Summary Count)
    table.Configure(options =>
    {
        options.OutputTo = testWriter;
        options.NumberAlignment = Alignment.Right;
        options.EnableCount = true;
    });

    // 4. Populate Workflow Data
    table.AddRow("Widget A", 150);
    table.AddRow("Gadget B", 9);

    // 5. Test Default Formatting and Count Visibility (Rule 1, Rule 11)
    table.Write(Format.Default);
    var defaultOutput = testWriter.ToString();
    Assert.Contains("Item", defaultOutput);
    Assert.Contains("Quantity", defaultOutput);
    Assert.Contains("Count: 2", defaultOutput);
    // Ensure Right Alignment is applied for numbers (Padding spaces before value)
    Assert.Contains("  150 |", defaultOutput);

    // 6. Test Format Switching - Minimal Style (Rule 1)
    testWriter.GetStringBuilder().Clear();
    table.Write(Format.Minimal);
    var minimalOutput = testWriter.ToString();
    Assert.DoesNotContain("|", minimalOutput);
    Assert.Contains("Widget A", minimalOutput);

    // 7. Test Format Switching - Markdown Style (Rule 10)
    testWriter.GetStringBuilder().Clear();
    table.Write(Format.MarkDown);
    var markdownOutput = testWriter.ToString();
    Assert.Contains("| Item", markdownOutput);
    Assert.Contains("|---", markdownOutput);

    // 8. Test Output Control - Footer Exclusion (Rule 6)
    testWriter.GetStringBuilder().Clear();
    table.Configure(o => o.EnableCount = false);
    table.Write(Format.Default);
    var outputWithoutCount = testWriter.ToString();
    Assert.DoesNotContain("Count: 2", outputWithoutCount);

    // 9. Verify Alternative Style (Rule 1)
    testWriter.GetStringBuilder().Clear();
    table.Write(Format.Alternative);
    var alternativeOutput = testWriter.ToString();
    Assert.Contains("+", alternativeOutput);
    Assert.Contains("Gadget B", alternativeOutput);
}
