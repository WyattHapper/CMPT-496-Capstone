// Workflow: Data Ingestion and Structural Validation
// Description: Handles the extraction of data from various sources like POCOs, dictionaries, and arrays while ensuring the integrity of column and row counts.
// Covered Business Rules: [2, 3, 5, 9]

using System;
using System.Collections.Generic;
using System.Data;
using System.Linq;
using Xunit;
using ConsoleTables;
using ConsoleTables.Tests;

[Fact]
public void DataIngestionAndStructuralValidation_IntegrationTest()
{
    // 1. Verify Ingestion from POCOs (Rule 2, 9)
    var users = new List<User>
    {
        new User { Name = "IntegrationTestUser", Age = 25 }
    };
    var pocoTable = ConsoleTable.From(users);
    Assert.Equal(2, pocoTable.Columns.Count);
    Assert.Contains("Name", pocoTable.Columns);
    Assert.Contains("Age", pocoTable.Columns);
    Assert.Single(pocoTable.Rows);
    Assert.Equal("IntegrationTestUser", pocoTable.Rows[0][0]);

    // 2. Verify Ingestion from Dictionaries with row labels (Rule 5, 9)
    var dictData = new Dictionary<string, Dictionary<string, object>>
    {
        { "RowA", new Dictionary<string, object> { { "Col1", true }, { "Col2", 50 } } }
    };
    var dictTable = ConsoleTable.FromDictionary(dictData);
    // Resulting columns: "" (label), "Col1", "Col2"
    Assert.Equal(3, dictTable.Columns.Count);
    Assert.Equal("", dictTable.Columns[0]);
    Assert.Equal("RowA", dictTable.Rows[0][0]);

    // 3. Verify Ingestion from DataTables and Object Arrays (Rule 9)
    var dt = new DataTable();
    dt.Columns.Add("TestCol");
    dt.Rows.Add("TestData");
    var dtTable = ConsoleTable.From(dt);
    Assert.Single(dtTable.Columns);
    Assert.Equal("TestData", dtTable.Rows[0][0]);

    Object[][] arrayData = 
    [
        ["H1", "H2"],
        ["V1", "V2"]
    ];
    var arrayTable = ConsoleTable.From(arrayData);
    Assert.Equal(2, arrayTable.Columns.Count);
    Assert.Equal("V1", arrayTable.Rows[0][0]);

    // 4. Verify Structural Integrity Rules (Rule 3)
    // Case A: Row addition fails if no columns are defined
    var emptyTable = new ConsoleTable(new ConsoleTableOptions { Columns = new List<string>() });
    var colEx = Assert.Throws<Exception>(() => emptyTable.AddRow("Value"));
    Assert.Equal("Please set the columns first", colEx.Message);

    // Case B: Row addition fails if column count mismatch
    var structuredTable = new ConsoleTable("Col1", "Col2");
    var mismatchEx = Assert.Throws<Exception>(() => structuredTable.AddRow("OnlyOne"));
    Assert.Contains("The number columns in the row (2) does not match the values (1)", mismatchEx.Message);

    // Case C: Row addition succeeds with correct count
    structuredTable.AddRow("Val1", "Val2");
    Assert.Single(structuredTable.Rows);
}

// Workflow: Visual Layout and Alignment Calculation
// Description: Processes cell content to determine column widths, handling Unicode character widths and applying specific alignment policies for numeric data types.
// Covered Business Rules: [4, 7, 8]

using System;
using System.Collections.Generic;
using System.Linq;
using Xunit;
using ConsoleTables;

[Fact]
public void VisualLayoutAndAlignment_ShouldHandleNumericAlignmentUnicodeWidthAndWordWrapping()
{
    // Arrange: Create a table with mixed data types to exercise alignment and layout rules
    var table = new ConsoleTable("ID", "Product", "Description");

    // Rule 7: Configure right-alignment for numeric data types
    table.Configure(o => o.NumberAlignment = Alignment.Right);

    // Rule 4: Set a MaxWidth and WordBreakDelimiter for wrapping logic
    table.MaxWidth = 15;
    table.SetWordBreakDelimiter(' ');

    // Rule 8: Use Unicode characters to verify visual width calculation
    // ID: 1234 (Numeric, should be right-aligned)
    // Product: "Table 哈哈" (Unicode characters, should calculate width correctly)
    // Description: A long string that exceeds MaxWidth (15) and should wrap at ' '
    table.AddRow(1234, "Table 哈哈", "This description is quite long and should wrap");

    // Act: Render the table to string
    var result = table.ToString();

    // Assert: Verify alignment, Unicode handling, and layout consistency

    // 1. Verify Numeric Right Alignment (Rule 7/8)
    // In a column of width 4 ("1234" vs "ID"), right alignment for "1234" 
    // results in padding on the left if the column were wider, but here it matches width.
    // We verify the numeric string is present in the output.
    Assert.Contains("| 1234 |", result);

    // 2. Verify Unicode Visual Width (Rule 4/8)
    // "Table 哈哈" contains two CJK characters. UnicodeCalculator.GetWidth typically returns 2 for each.
    // Visual width = 6 (Table ) + 4 (哈哈) = 10.
    // The table rendering should account for this to keep vertical bars | aligned.
    Assert.Contains("Table 哈哈", result);

    // 3. Verify Layout Wrapping (Rule 4)
    // The description "This description is quite long and should wrap" (46 chars) 
    // should be wrapped because MaxWidth is 15.
    // The first segment should be cut at the word break delimiter ' '.
    // "This " (5), "description " (12) -> "This description" is 16 chars (exceeds 15).
    // So it should cut at the space before "description" or after "This".
    // The SetFormats logic: cutCell = cellStr[..15] -> "This descriptio". 
    // Since last char is not space, it looks for lastIndexOf(' ') in cutCell.
    // "This " is found at index 4. So cutCell becomes "This".
    Assert.Contains("| This          |", result);
    Assert.Contains("| description   |", result);
    Assert.Contains("| is quite long |", result);

    // 4. Verify Count Header/Footer (General Layout)
    Assert.Contains("Count: 1", result);
}

// Workflow: Tabular Rendering and Output Execution
// Description: Applies visual styles such as Markdown or Minimal to the calculated layout and renders the final table to the system output with optional metadata like row counts.
// Covered Business Rules: [1, 6, 10, 11]

using System;
using System.Collections.Generic;
using System.IO;
using Xunit;
using ConsoleTables;
using ConsoleTables.Tests;

[Fact]
public void TabularRenderingAndOutputExecution_WorkflowTest()
{
    // Step 1: Verify Default System Output (Rule 10)
    var defaultTable = new ConsoleTable("test");
    Assert.Equal(Console.Out, defaultTable.Options.OutputTo);

    // Step 2: Prepare data using the existing User model from the test suite
    var users = new List<User>
    {
        new User { Name = "Integration User", Age = 30 }
    };

    using (var writer = new StringWriter())
    {
        // Step 3: Setup ConsoleTable with custom configuration (Rules 1, 10)
        // We configure Right alignment for numeric columns (Rule 1) and redirect output (Rule 10)
        var table = ConsoleTable.From(users)
            .Configure(o =>
            {
                o.OutputTo = writer;
                o.NumberAlignment = Alignment.Right;
                o.EnableCount = true;
            });

        // Step 4: Execute Markdown rendering (Rule 1, Rule 10)
        table.Write(Format.MarkDown);
        var markdownOutput = writer.ToString();

        // Verification: Check Markdown syntax and Right Alignment for Age column (Rule 1)
        Assert.Contains("| Name             | Age |", markdownOutput);
        Assert.Contains("| Integration User |  30 |", markdownOutput);

        // Step 5: Execute Default rendering and verify Row Count (Rule 11)
        writer.GetStringBuilder().Clear();
        table.Write(Format.Default);
        var defaultOutput = writer.ToString();

        Assert.Contains("Count: 1", defaultOutput);

        // Step 6: Exclude row count and verify output (Rule 6)
        writer.GetStringBuilder().Clear();
        table.Configure(o => o.EnableCount = false);
        table.Write(Format.Default);
        var noCountOutput = writer.ToString();

        Assert.Contains("Integration User", noCountOutput);
        Assert.DoesNotContain("Count:", noCountOutput);

        // Step 7: Verify Minimal style formatting (Rule 1, 10)
        writer.GetStringBuilder().Clear();
        table.Write(Format.Minimal);
        var minimalOutput = writer.ToString();

        Assert.Contains("Name", minimalOutput);
        Assert.Contains("Integration User", minimalOutput);
        // Minimal format typically removes pipe separators
        Assert.DoesNotContain("|", minimalOutput);
    }
}
