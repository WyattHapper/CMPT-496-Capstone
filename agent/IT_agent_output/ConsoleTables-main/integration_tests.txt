// Workflow: Visual Layout and Alignment Configuration
// Description: Defines the rules for positioning data within columns, including handling multi-byte Unicode characters, numeric alignment, and word wrapping.
// Covered Business Rules: [4, 7, 8]

using System;
using System.Collections.Generic;
using System.IO;
using Xunit;
using ConsoleTables;
using ConsoleTables.Tests;

[Fact]
public void VisualLayoutAndAlignment_ShouldRespectUnicodeAndNumericConfigurations()
{
    // Arrange: Create a data set using an existing application model (User from ConsoleTableTest)
    var users = new List<User>
    {
        // Rule 8: Multi-byte Unicode characters ("哈哈" = visual width 4)
        new User { Name = "User 哈哈", Age = 25 },
        // Rule 7/8: Right alignment for numeric data types
        new User { Name = "Short", Age = 1000 }
    };

    var output = new StringWriter();

    // Act: Initialize the table and apply layout configurations
    var table = ConsoleTable.From(users);
    
    // Rule 4: Configure wrapping parameters
    table.MaxWidth = 10;
    table.SetWordBreakDelimiter(' ');
    
    // Rule 7: Configure numeric alignment policy
    table.Configure(o => 
    {
        o.NumberAlignment = Alignment.Right;
        o.OutputTo = output;
        o.EnableCount = true;
    });

    // Render the table
    table.Write();

    var result = output.ToString();

    // Assert: Verify visual alignment and numeric right-alignment
    // The Age column header is length 3. "25" and "1000" should be padded correctly.
    // Since Age 1000 is 4 chars, and the column must be at least 4 wide (header 'Age' is 3),
    // Age 1000 will define width as 4. Age 25 should have two leading spaces to right-align.
    
    // Assert Unicode presence (Rule 8)
    Assert.Contains("User 哈哈", result);

    // Assert Numeric Right Alignment (Rule 7)
    // The Age column for 25 should be right-aligned. Based on code logic, 
    // format string {i, length} without '-' results in right alignment.
    Assert.Contains("|   25 |", result);
    Assert.Contains("| 1000 |", result);

    // Rule 4: Verify MaxWidth/Wrapping property exists and was applied to configuration logic
    Assert.Equal(10, table.MaxWidth);
    Assert.Equal(' ', table.WordBreakDelimiter);
    
    // Rule 8: Confirm visual layout maintained count
    Assert.Contains("Count: 2", result);
}

// Workflow: Table Rendering and Output
// Description: Produces the final visual representation of the table using specific styles and formats, managing output destination and optional footer metadata.
// Covered Business Rules: [1, 6, 10, 11]

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using Xunit;
using ConsoleTables;
using ConsoleTables.Tests;

[Fact]
public void TableRendering_FullWorkflow_ShouldApplyStylesAndManageMetadata()
{
    // Arrange: Create test data using the existing User entity found in ConsoleTableTest.cs
    var users = new List<User>
    {
        new User { Name = "Alexandre", Age = 36 },
        new User { Name = "René", Age = 59 }
    };

    var writer = new StringWriter();
    
    // Act: Build table from POCO list and configure options
    var table = ConsoleTable.From(users);
    
    // Rule 1 & 10: Configure Alignment and custom OutputTo destination
    table.Configure(o => 
    {
        o.NumberAlignment = Alignment.Right; 
        o.OutputTo = writer;
        o.EnableCount = true; // Rule 11: Summary count
    });

    // Act - Execution 1: Render using Default format (Rule 1 & 10)
    table.Write(Format.Default);
    var defaultOutput = writer.ToString();

    // Assertions for Execution 1
    // Rule 1: Verify Right Alignment for numeric type 'Age'
    Assert.Contains(" Age |", defaultOutput); 
    Assert.Contains("  36 |", defaultOutput);
    Assert.Contains("  59 |", defaultOutput);
    // Rule 11: Verify total row count processed
    Assert.Contains("Count: 2", defaultOutput);

    // Act - Execution 2: Render using Alternative style and exclude count (Rule 1 & 6)
    writer.GetStringBuilder().Clear();
    table.Configure(o => o.EnableCount = false);
    table.Write(Format.Alternative);
    var altOutput = writer.ToString();

    // Assertions for Execution 2
    // Rule 1: Verify Alternative visual style (uses + and - separators)
    Assert.Contains("+", altOutput);
    Assert.Contains("-", altOutput);
    // Rule 6: Verify exclusion of total row count from output
    Assert.DoesNotContain("Count:", altOutput);

    // Act - Execution 3: Render using Minimal format (Rule 10)
    writer.GetStringBuilder().Clear();
    table.Write(Format.Minimal);
    var minimalOutput = writer.ToString();

    // Assertions for Execution 3
    // Rule 10: Verify Minimal format structure
    Assert.Contains("Name", minimalOutput);
    Assert.Contains("Age", minimalOutput);
    Assert.DoesNotContain("|", minimalOutput);
}
